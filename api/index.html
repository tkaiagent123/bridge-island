class Datum:
    """One element value with its *source* time of validity (G3), never delivery time.

    An *estimated* datum (see the Bridge's ``extrapolate`` rate policy) is a prediction, not a
    measurement. Its ``source_time`` is the time the estimate is valid for, and ``basis_time``
    is the time of the last real sample it was predicted from. Its age is measured from the
    basis, so an estimate can never make old information look fresh.
    """

    element: str
    value: Any
    source_time: float
    estimated: bool = False
    basis_time: float | None = None

    def age(self, now: float) -> float:
        basis = (
            self.basis_time if self.estimated and self.basis_time is not None else self.source_time
        )
        return now - basis

class BamDecode(Transform):
    """Binary angular measurement counts to degrees (float)."""

    bits: int = 16

    @property
    def _step(self) -> float:
        return 360.0 / (1 << self.bits)

    def apply(self, value: int) -> float:
        return value * self._step

    def expr(self, x: str) -> str:
        return f"({x} * {self._step!r})"

    def inverse(self) -> Transform:
        return BamEncode(self.bits)

    def output(self, rep: Representation) -> Representation:
        if rep.encoding != f"bam{self.bits}":
            raise TransformError(f"expected encoding bam{self.bits}, got {rep.encoding!r}")
        return replace(rep, encoding="float64", units="deg")

class BitUnpack(Transform):
    """Integer word to a dict of named bit fields (least-significant field first)."""

    fields: tuple[tuple[str, int], ...]

    def apply(self, value: int) -> dict[str, int]:
        return {n: (value >> s) & m for n, s, m in _layout(self.fields)}

    def expr(self, x: str) -> str:
        items = ", ".join(f"{n!r}: ({x} >> {s}) & {m}" for n, s, m in _layout(self.fields))
        return "{" + items + "}"

    def inverse(self) -> Transform:
        return BitPack(self.fields)

    def output(self, rep: Representation) -> Representation:
        return replace(rep, encoding="bitfield")

    def check_type(value: Any, encoding: str) -> None:
    """Shallow type check of a value against an encoding name."""
    if encoding.startswith(("float",)):
        ok = isinstance(value, (int, float)) and not isinstance(value, bool)
    elif encoding.startswith(("uint", "bam")):
        bits = int("".join(c for c in encoding if c.isdigit()) or 0)
        ok = isinstance(value, int) and not isinstance(value, bool)
        ok = ok and (bits == 0 or 0 <= value < 2**bits)
    elif encoding == "bool":
        ok = isinstance(value, bool)
    else:
        ok = True
    if not ok:
        raise MalformedError(f"{value!r} is not a valid {encoding}")

    def from_spec(spec: Any) -> Transform:
    """Build a transform from one YAML mapping-spec entry.

    Raises :class:`TransformError`, and nothing else, for anything that is not a permitted,
    well-formed transform.
    """
    if isinstance(spec, str) and spec == "identity":
        return Identity()
    if isinstance(spec, dict) and len(spec) == 1:
        ((kind, arg),) = spec.items()
        builder = _BUILDERS.get(kind) if isinstance(kind, str) else None
        if builder is not None:
            return builder(arg)
    raise TransformError(f"transform not permitted: {spec!r}")

    def coefficients(src: str, dst: str) -> tuple[float, float]:
    """Return ``(scale, offset)`` such that ``dst_value = src_value * scale + offset``."""
    if src == dst:
        _lookup(src)
        return 1.0, 0.0
    dim_s, scale_s, off_s = _lookup(src)
    dim_d, scale_d, off_d = _lookup(dst)
    if dim_s != dim_d:
        raise UnitError(f"cannot convert {src!r} ({dim_s}) to {dst!r} ({dim_d})")
    return scale_s / scale_d, (off_s - off_d) / scale_d

    class Scheduler:
    """Runs callbacks in simulated-time order. Ties run in scheduling order."""

    def __init__(self, seed: int = 0) -> None:
        self.now = 0.0
        self.rng = random.Random(seed)
        self._queue: list[tuple[float, int, Callable[[], None]]] = []
        self._seq = 0

    def at(self, time: float, fn: Callable[[], None]) -> None:
        time = round(time, 9)  # keeps k * period free of float drift
        if time < self.now:
            raise ValueError(f"cannot schedule at {time} before now={self.now}")
        heapq.heappush(self._queue, (time, self._seq, fn))
        self._seq += 1

    def after(self, delay: float, fn: Callable[[], None]) -> None:
        self.at(self.now + delay, fn)

    def every(self, period: float, fn: Callable[[], None], start: float = 0.0) -> None:
        """Call ``fn`` at ``start + k * period`` for k = 0, 1, 2, ..."""

        def tick(k: int) -> None:
            fn()
            self.at(start + (k + 1) * period, lambda: tick(k + 1))

        self.at(start, lambda: tick(0))

    def run_until(self, end: float) -> None:
        while self._queue and self._queue[0][0] <= end:
            time, _, fn = heapq.heappop(self._queue)
            self.now = time
            fn()
        self.now = max(self.now, end)

    class Network:
    """Registry of nodes and the links wired between them."""

    def __init__(self) -> None:
        self.islands: dict[str, str] = {}
        self.links: list[Link] = []

    def add(self, member: IslandMember) -> None:
        self.islands[member.name] = member.island

    def connect(self, a: str, b: str, protocol: str) -> None:
        for n in (a, b):
            if n not in self.islands:
                raise KeyError(f"unknown node {n!r}")
        self.links.append(Link(a, b, protocol))

    def bypass_links(self) -> list[Link]:
        """Links that join a baseline node directly to a unique node."""
        return [
            link
            for link in self.links
            if {self.islands[link.a], self.islands[link.b]} == {"baseline", "unique"}
        ]

    def problems(self) -> list[str]:
        """G1 violations: bypass links, and unique nodes with no link to the Bridge."""
        found = [f"bypass link {lk.a} <-> {lk.b}" for lk in self.bypass_links()]
        for name, island in self.islands.items():
            if island != "unique":
                continue
            peers = {lk.b if lk.a == name else lk.a for lk in self.links if name in (lk.a, lk.b)}
            if not any(self.islands[p] == "bridge" for p in peers):
                found.append(f"{name} has no link to the Bridge")
        return found

        self.transfers_done = 0

    def every_for(self, rate_hz: float) -> int:
        """Minor frames between transfers for a rate; the rate must divide the minor rate."""
        ratio = self.minor_hz / rate_hz
        if abs(ratio - round(ratio)) > 1e-9 or ratio < 1:
            raise BusError(f"{rate_hz} Hz does not divide the {self.minor_hz} Hz minor frame")
        return round(ratio)

    def set_controller(self, bc: object) -> None:
        if not isinstance(bc, BusController):
            raise BusError("only a BusController can control the bus; a remote terminal cannot")
        if self.controller is not None:
            raise BusError("bus already has a controller")
        self.controller = bc

    def attach(self, rt: RemoteTerminal) -> int:
        if len(self.terminals) >= 30:
            raise BusError("a bus has at most 30 remote terminals")
        rt.address = len(self.terminals) + 1
        self.terminals[rt.address] = rt
        return rt.address

    def start(self) -> None:
        if self.controller is None:
            raise BusError("no bus controller")
        frame = [0]

        def minor() -> None:
            assert self.controller is not None
            for t in self.controller.transfers:
                if frame[0] % t.every == t.offset:
                    self._run(t)
            frame[0] += 1

        self.sched.every(1.0 / self.minor_hz, minor)

    def _run(self, t: Transfer) -> None:
        rt = self.terminals.get(t.rt)
        if rt is None:
            self.errors.append(f"no response from RT {t.rt}")
            return
        self.transfers_done += 1
        if t.direction == "bc_to_rt":
            msg = t.provide() if t.provide else None
            handler = rt.on_receive.get(t.subaddress)
            if msg is None:
                return
            if handler is None:
                self.errors.append(f"RT {t.rt} has no receive subaddress {t.subaddress}")
                return
            self.sched.after(self.latency_s, lambda: handler(msg))
        else:
            source = rt.on_transmit.get(t.subaddress)
            if source is None:
                self.errors.append(f"RT {t.rt} has no transmit subaddress {t.subaddress}")
                return
            reply = source()
            consume = t.consume
            if consume:
                self.sched.after(self.latency_s, lambda: consume(reply))

    def connect(self, rt: RemoteTerminal, owner: IslandMember) -> int:
        """Attach a remote terminal. Returns its bus address."""
        if self.enforce_bridge and owner.island != "bridge":
            raise BypassError(f"{owner.name!r} ({owner.island}) may not attach to the baseline")
        return self.bus.attach(rt)

    def start(self) -> None:
        if self._started:
            return
        self._started = True
        for rt_addr in self.bus.terminals:
            rt = self.bus.terminals[rt_addr]
            for spec in self.interface.published:
                if spec.subaddress not in rt.on_receive:
                    continue  # this terminal does not take that message
                self._bc.transfers.append(
                    Transfer(
                        rt_addr,
                        spec.subaddress,
                        "bc_to_rt",
                        every=self.bus.every_for(spec.rate_hz),
                        provide=lambda spec=spec: self._emit(spec),  # type: ignore[misc]
                    )
                )
            for spec in self.interface.accepted:
                if spec.subaddress not in rt.on_transmit:
                    continue
                self._bc.transfers.append(
                    Transfer(
                        rt_addr,
                        spec.subaddress,
                        "rt_to_bc",
                        every=self.bus.every_for(spec.rate_hz),
                        consume=self._accept,
                    )
                )
        self.bus.start()

    def _emit(self, spec: MessageSpec) -> Message:
        valid_at = round(self.sched.now - self.sample_age_s, 9)
        self.emitted[spec.name] = self.emitted.get(spec.name, 0) + 1
        return Message(spec.name, dict(self.sources[spec.name](valid_at)), valid_at)

    def _accept(self, msg: Message | None) -> None:
        if msg is None:
            return
        now = self.sched.now
        spec = self.interface.accepted_message(msg.spec_name)
        if spec is None:
            self.rejected.append((now, f"{msg.spec_name} is not an accepted input"))
        elif set(msg.values) != {e.name for e in spec.elements}:
            self.rejected.append((now, f"{msg.spec_name} has the wrong fields"))
        else:

                v = msg.values[e.name]
                if e.valid_range and not (
                    isinstance(v, (int, float)) and e.valid_range[0] <= v <= e.valid_range[1]

                    self.rejected.append((now, f"{msg.spec_name}.{e.name}={v!r} out of range"))
                    return
            self.accepted.append((now, msg))

        rec = Received(native_name, datum, self.sched.now)
        self.received.append(rec)
        self._latest[native_name] = rec
        need = self.need(native_name)
        if need and rec.age > need.latency_budget_s:
            self.late.append(rec)

        self.unavailable[native_name] = reason

    def on_annunciation(self, kind: str, detail: str) -> None:
        self.annunciations.append((self.sched.now, kind, detail))

    def latest(self, native_name: str) -> Received | None:
        return self._latest.get(native_name)

    def check(self) -> list[str]:
        """Consumer-side checks at the current time: missing, unavailable or stale inputs."""
        issues = []
        for need in self.needs:
            if need.native_name in self.unavailable:
                issues.append(
                    f"{need.native_name}: unavailable ({self.unavailable[need.native_name]})"
                )

            if last is None:

        """Send a message toward the baseline. The Bridge, not the sender, validates it."""
        if self._uplink is None:
            raise RuntimeError(f"{self.name} is not connected to a Bridge")
        self._uplink(Message(message_name, dict(values), self.sched.now))

    def _check_fit(self) -> None:
        """G4: consume only what the baseline publishes, emit only what it accepts."""
        m = self.mapping
        problems = []
        if self.interface.version != m.BASELINE_VERSION:
            problems.append(
                f"mapping is for baseline v{m.BASELINE_VERSION}, not v{self.interface.version}"
            )
        problems += [
            f"{ref} is not published by baseline v{self.interface.version}"
            for ref, _, _, _ in m.DOWNLINKS
            if self.interface.element(ref) is None
        ]
        problems += [
            f"{msg} is not accepted by baseline v{self.interface.version}"
            for _, _, msg in m.UPLINKS
            if self.interface.accepted_message(msg) is None
        ]
        if problems:

    def attach_baseline(self, platform: BaselinePlatform) -> None:
        platform.connect(self.oiu.rt, owner=self)

    def start(self) -> None:
        """Report removed baseline elements to the affected subsystems, and start monitoring."""
        if self.oiu.rt.address is None:
            raise InterfaceMismatch("no baseline is attached: call attach_baseline() first")
        mapped = (
            {s for _, s, _, _ in self.mapping.DOWNLINKS}
            | {s for _, s, _, _ in self.mapping.UNAVAILABLE}
            | {s for s, _, _ in self.mapping.UPLINKS}
        )
        missing = sorted(mapped - set(self.uid.subsystems))
        if missing:  # otherwise the first datum for them would crash the run, much later
            raise InterfaceMismatch(
                f"the mapping refers to subsystems that are not attached: {missing}"
            )
        for ref, sub, native, reason in self.mapping.UNAVAILABLE:
            self.log.add(
                self.sched.now,
                self.name,
                "bridge.unavailable",
                f"{ref} -> {sub}.{native}: {reason}",
            )
            if sub in self.uid.subsystems:
                self.uid.deliver_unavailable(sub, native, reason)
        period = self.health.check_period_s
        self.sched.every(period, self._scan_if_running, start=period)

    def _scan_if_running(self) -> None:
        if not self.stalled:
            self.health.scan()

    def stall(self, at: float, duration: float) -> None:
        """Fault injection: the Bridge stops processing at ``at`` and resumes after ``duration``."""
        self.sched.at(at, self._begin_stall)
        self.sched.at(at + duration, self._end_stall)

    def _begin_stall(self) -> None:
        self.stalled = True
        self.log.add(self.sched.now, self.name, "bridge.stall", "processing stopped")

    def _end_stall(self) -> None:
        self.stalled = False
        self.log.add(self.sched.now, self.name, "bridge.resume", "processing resumed")
        held, self._held = self._held, {}
        for msg in held.values():  # the RT buffer keeps only the latest message per subaddress
            self._on_baseline(msg)

    # --- data paths -----------------------------------------------------------------

    def _hold(self, msg: Message) -> None:
        """While stalled, keep only the latest message per name, like an RT subaddress buffer."""
        held = self._held.get(msg.spec_name)
        if held is None or msg.source_time >= held.source_time:
            self._held[msg.spec_name] = msg

    def _on_baseline(self, msg: Message) -> None:
        if self.stalled:
            self._hold(msg)
            return
        self.health.observe(msg)
        self.sched.after(self.processing_delay_s, lambda: self._forward(msg))

    def _forward(self, msg: Message) -> None:
        if self.stalled:  # in flight when the stall began: released, still old, on resume
            self._hold(msg)
            return
        for d in self.tm.down(msg):
            if self.cyber.check_down(d.ref, d.subsystem):
                self.uid.deliver(d.subsystem, d.datum)

    def _on_unique(self, sender: str, msg: Message) -> None:
        if self.stalled:
            self.log.add(
                self.sched.now,
                self.name,
                "bridge.drop_up",
                f"{sender}.{msg.spec_name} while stalled",
            )
            return
        if not self.cyber.check_up(sender, msg.spec_name):
            return
        try:
            out = self.tm.up(sender, msg)
        except TransformError as exc:
            self.tm.rejected += 1
            self.log.add(self.sched.now, "tm", "tm.reject_up", f"{sender}.{msg.spec_name}: {exc}")

        baseline supplies: ``"cap"`` delivers only real samples, at most at the subsystem's rate;

        self._gridded: set[tuple[str, str]] = set()
        self._ticking: set[tuple[str, str]] = set()
        self.sent = 0
        self.decimated = 0
        self.estimates = 0

    def connect(self, sub: UniqueSubsystem) -> None:
        if sub.protocol not in _PORTS:
            raise ValueError(f"{sub.name}: unknown native protocol {sub.protocol!r}")
        self.subsystems[sub.name] = sub
        self._ports[sub.name] = _PORTS[sub.protocol](self.sched, sub, self.on_uplink)

    def deliver(self, subsystem: str, datum: Datum) -> bool:
        """Send a datum on the subsystem's native connection, capped at its useful rate."""
        need = self.subsystems[subsystem].need(datum.element)
        if need is None:
            raise KeyError(f"{subsystem} has no input {datum.element!r}")
        key = (subsystem, datum.element)
        history = self._history.setdefault(key, deque(maxlen=2))
        history.append((datum.source_time, datum.value))
        period = 1.0 / need.rate_hz
        slow = len(history) == 2 and history[1][0] - history[0][0] > period * 1.01
        if self.rate_policy == "extrapolate" and slow:
            # The source is slower than the subsystem can use: deliver on the subsystem's own
            # rate grid, one slot per period, carrying the freshest real sample if one has
            # arrived and an estimate otherwise. Real samples wait for their slot.
            self._gridded.add(key)
            self._pending[key] = datum
            if key not in self._ticking:
                self._ticking.add(key)
                self.sched.every(period, lambda: self._tick(key), start=self.sched.now + period)
            return True
        self._gridded.discard(key)
        self._pending.pop(key, None)
        # The cap is measured in *source* time: it limits the rate of the data, not of its
        # arrival. (Measured in arrival time, a message released late after a stall would use up
        # the slot and a fresh one 0.3 s later would be dropped.) If source time goes backwards,
        # for example after a source restart, accept the datum and start counting again.
        last = self._last_sent.get(key)
        if last is not None and 0 <= datum.source_time - last < period - 1e-6:
            self.decimated += 1
            return False
        self._last_sent[key] = datum.source_time
        self._ports[subsystem].send(datum.element, datum)
        self.sent += 1
        return True

    def _tick(self, key: tuple[str, str]) -> None:
        """One slot of the subsystem's rate grid: a real sample if pending, else an estimate."""
        if key not in self._gridded or not self._is_running():
            return
        subsystem, native = key
        pending = self._pending.pop(key, None)
        if pending is not None:
            self._ports[subsystem].send(native, pending)
            self.sent += 1
            return
        need = self.subsystems[subsystem].need(native)
        if need is None:
            return
        history = self._history[key]
        now = self.sched.now
        ok, value = estimate(
            need.representation.encoding,
            need.representation.units,
            list(history),
            now,
            self.max_extrapolation_s,
        )
        if ok:
            self.estimates += 1
            self._ports[subsystem].send(native, Datum(native, value, now, True, history[-1][0]))

    def estimate(
    encoding: str,
    units: str,
    history: list[tuple[float, Any]],
    now: float,
    limit: float | None = None,
) -> tuple[bool, Any]:
    """Estimate a value at ``now`` from the last one or two real samples ``(time, value)``.

    Returns ``(True, value)`` or ``(False, None)`` when no estimate should be made. Floats are
    extrapolated linearly, other encodings (integer codes, enums, booleans) are held. No estimate
    is made later than ``limit`` seconds after the last real sample (default: 1.5 source
    periods), so a dead source is not masked, nor across what looks like an angle wrapping
    round, where a straight line would be badly wrong.
    """
    t1, v1 = history[-1]
    if limit is None:
        limit = 1.5 * (t1 - history[0][0]) if len(history) == 2 else 0.0
    if now - t1 > limit:
        return False, None
    if not encoding.startswith("float"):
        return True, v1
    if len(history) < 2 or t1 <= history[0][0]:
        return False, None
    t0, v0 = history[0]
    half_turn = _HALF_TURN.get(units)
    if half_turn is not None and abs(v1 - v0) > half_turn:
        return False, None
    return True, v1 + (v1 - v0) / (t1 - t0) * (now - t1)

    class DirectAdapter:
    """A subsystem's own baseline adapter. ``mapping`` is the generated module it was written
    against, and ``interface`` the baseline version that module targets."""

    def __init__(
        self,
        platform: BaselinePlatform,
        sub: UniqueSubsystem,
        mapping: ModuleType,
        interface: BaselineInterface,
    ) -> None:
        self.sub, self.mapping = sub, mapping
        self.rt = RemoteTerminal(sub.name)
        self.errors: list[tuple[float, str, str]] = []
        self.uplink_queue: deque[Message] = deque()
        self._last_sent: dict[str, float] = {}
        self._refs = {ref for ref, s, _, _ in mapping.DOWNLINKS if s == sub.name}
        for message in {ref.split(".")[0] for ref in self._refs}:
            spec = interface.published_message(message)
            if spec is not None:
                self.rt.on_receive[spec.subaddress] = self._on_message
        for _, s, native, reason in mapping.UNAVAILABLE:
            if s == sub.name:  # the subsystem's own developers had to add this handling
                sub.on_unavailable(native, reason)
        if sub.outputs:
            sub._uplink = self._send
        for spec in interface.accepted:  # every terminal answers the baseline's polls
            self.rt.on_transmit[spec.subaddress] = self._next_uplink
        platform.connect(self.rt, owner=sub)

    def _on_message(self, msg: Message) -> None:
        now = self.sub.sched.now
        outs, errors = self.mapping.translate(msg.spec_name, msg.values)
        for ref, why in errors:
            if ref in self._refs:
                self.errors.append((now, ref, why))
        for _, s, native, value, _ in outs:
            need = self.sub.need(native)
            if s != self.sub.name or need is None:
                continue
            last = self._last_sent.get(native)
            if last is not None and now - last < 1.0 / need.rate_hz - 1e-9:
                continue
            self._last_sent[native] = now
            self.sub.on_data(native, Datum(native, value, msg.source_time))

    def _send(self, msg: Message) -> None:
        try:
            name, values = self.mapping.translate_up(self.sub.name, msg.spec_name, msg.values)
        except TransformError as exc:  # all the protection a P2P subsystem has is its own code
            self.errors.append((self.sub.sched.now, msg.spec_name, str(exc)))
            return
        self.uplink_queue.append(Message(name, values, msg.source_time))

    def send_raw(self, name: str, values: dict[str, Any]) -> None:
        """What faulty subsystem software can do in P2P: put anything on the baseline bus."""
        self.uplink_queue.append(Message(name, values, self.sub.sched.now))

    def _next_uplink(self) -> Message | None:
        return self.uplink_queue.popleft() if self.uplink_queue else None

    What a baseline interface change touches, in each architecture.

    Counting model (stated assumptions, not measurements): * The unit of counting is the element-level dependency: an edge (baseline data element, subsystem). A count taken at another level, for example baseline systems or units, is a different quantity and is not comparable with these. * Every dependency edge has tests_per_edge conformance tests (default 1), located in whatever component owns the adapter for that edge. Every subsystem also has one acceptance test. The value of tests_per_edge is arbitrary and it alone drives the size of the "tests failing" row; the difference between the architectures in that row does not depend on it (it is the number of P2P subsystems with code changes minus the number of Bridge subsystems with code changes). * P2P: each subsystem owns its own adapter module. A rename, unit, encoding or removal change needs code in every adapter that reads the element. A rate change needs no code but invalidates timing assumptions, so those subsystems are re-verified. * Bridge: the adapters are entries in one mapping, so those changes touch the Translator Module only. A removed element is the exception: the affected subsystem must also handle the "unavailable" report. The Bridge itself is re-verified after any baseline change.

    def run_scenario(scenario: Scenario, world: ModuleType) -> tuple[System, int]:
    """Build the system, schedule the events and run it. Returns the system and the version."""
    version = scenario.version if scenario.version is not None else world.VERSIONS[0]
    if version not in world.VERSIONS:
        raise ScenarioError(
            f"{scenario.name}: the {scenario.example} example has baseline versions {list(world.VERSIONS)}, not {version}"
        )
    system = world.build(version, scenario.seed, **scenario.bridge)
    assert system.bridge is not None
    for i, event in enumerate(scenario.events):
        at = float(event["at"])
        if "stall" in event:
            system.bridge.stall(at, float(event["stall"]))
            continue
        send = event["send"]
        sender = system.subsystems.get(send["from"])
        if sender is None:
            raise ScenarioError(
                f"{scenario.name}: events[{i}].send.from {send['from']!r} is not a subsystem; "
                f"choose from {sorted(system.subsystems)}"
            )
        system.sched.at(
            at,
            lambda s=sender, m=send["message"], v=dict(send["values"]): s.send(m, v),
        )
    system.run(scenario.duration_s)
    return system, version

    """The whole app, with no sockets: ``handle`` takes a request and returns a response."""

    def __init__(self, port: int = 0) -> None:
        self.token = secrets.token_urlsafe(16)
        self.port = port
        self._results: OrderedDict[str, dict[str, tuple[str, bytes]]] = OrderedDict()
        self._lock = threading.Lock()

    # ---- plumbing ------------------------------------------------------------------------

    def _allowed_hosts(self) -> set[str]:
        return {f"127.0.0.1:{self.port}", f"localhost:{self.port}"}

    def _store(self, files: dict[str, tuple[str, bytes]]) -> str:
        run_id = secrets.token_urlsafe(9)
        self._results[run_id] = files
        while len(self._results) > KEEP_RESULTS:
            self._results.popitem(last=False)
        return run_id

    def handle(
        self, method: str, target: str, headers: Mapping[str, str], body: bytes = b""
    ) -> Response:
        if headers.get("Host", "") not in self._allowed_hosts():
            return self._plain(403, "Unexpected Host header. Open the address printed by the app.")
        url = urlsplit(target)
        query = parse_qs(url.query, keep_blank_values=True)
        try:
            if method == "GET":
                return self._get(url.path, query)
            if method == "POST":
                if len(body) > MAX_BODY:
                    return self._plain(413, "That request is too large.")
                form = parse_qs(body.decode("utf-8", "replace"), keep_blank_values=True)
                if not hmac.compare_digest(_first(form, "t"), self.token):
                    return self._plain(
                        403, "This page has expired. Go back, reload it and try again."
                    )
                return self._post(url.path, form)
        except ScenarioError as exc:
            return self._page("Something to fix", self._error_html(str(exc)), "/", status=200)
        except FileNotFoundError as exc:
            message = (
                "The example files were not found. Run the app from a clone of the repository, "
                f"or set BRIDGE_ISLAND_EXAMPLES. ({exc})"
            )
            return self._page("Examples not found", self._error_html(message), "/", status=200)
        except Exception:  # last line of defence: one bad request must not take the app down
            traceback.print_exc()
            message = "Something went wrong inside the app. The details are in the black window."
            return self._page("Something went wrong", self._error_html(message), "/", status=500)
        return self._plain(405, "Method not allowed.")

        return Response(status, text.encode("utf-8"), "text/plain; charset=utf-8")

        policy = _first(form, "rate_policy", "cap")
        if policy not in ("cap", "extrapolate"):
            raise ScenarioError("Choose a rate policy from the list.")
        stall = None
        if _first(form, "stall"):
            stall = (
                _num(form, "stall_at", 2.2, 0, duration, "Stall start"),
                _num(form, "stall_for", 1.5, 0.1, duration, "Stall length"),
            )
        with self._lock:
            page = build_report(load_world(example), seed, policy, duration, stall)
        if _first(form, "mode") == "download":
            return Response(
                200,
                page.encode(),
                headers={
                    "Content-Disposition": f'attachment; filename="bridge-island-{example}.html"'
                },
            )
        bar = '<p><a href="/report">&larr; Back to bridge-island</a></p>'
        return Response(200, page.replace('<main id="main">', f'<main id="main">{bar}', 1).encode())

    def _experiment(self, form: Mapping[str, list[str]]) -> Response:

        seed = int(_num(form, "seed", 1, -(10**9), 10**9, "Random seed"))
        trials = int(_num(form, "trials", 300, 10, MAX_TRIALS, "Random trials"))
        buffer = io.StringIO()
        with self._lock, tempfile.TemporaryDirectory() as tmp, contextlib.redirect_stdout(buffer):
            files = experiments.run(name, load_world(example), seed, Path(tmp), trials)
            stored: dict[str, tuple[str, bytes]] = {}
            images = []
            for path in files.values():
                data = path.read_bytes()
                if path.suffix == ".png":
                    images.append(
                        f'<img class="plot" alt="{E(path.stem.replace("_", " "))} (synthetic, illustrative)" src="data:image/png;base64,{base64.b64encode(data).decode()}">'
                    )
                stored[path.name] = (
                    "image/png" if path.suffix == ".png" else "text/csv; charset=utf-8",
                    data,
                )
        run_id = self._store(stored)
        links = "".join(
            f'<a class="button secondary" href="/download/{run_id}/{E(n)}">{E(n)}</a>'
            for n in stored
        )
        text = "\n".join(
            line for line in buffer.getvalue().splitlines() if not line.startswith("wrote ")
        )
        title = EXPERIMENTS[name][0]
        body = (
            f'<h1>{E(title)}: {E(example)}</h1><p class="banner">{E(NOTE)} Seed {seed}.</p>'
            f"<pre>{E(text)}</pre>{''.join(images)}<h2>Files</h2><p>{links}</p>"
            '<p><a class="button secondary" href="/experiments">Run another</a></p>'
        )
        return self._page(title, body, "/experiments")

    def make_server(port: int = 8765) -> tuple[ThreadingHTTPServer, App]:
    """Bind to 127.0.0.1 only, trying the next ports if the first is busy."""
    last: OSError | None = None
    for candidate in range(port, port + 20) if port else [0]:
        app = App(candidate)
        try:
            server = _Server(("127.0.0.1", candidate), make_handler(app))
        except OSError as exc:
            last = exc
            continue
        app.port = server.server_address[1]
        return server, app
    raise OSError(f"no free port from {port} to {port + 19}: {last}")
